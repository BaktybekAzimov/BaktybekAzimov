### Ошибка при FIX #2 (формат телефона)

**Ошибка:** `new row violates check constraint "check_phone_format"`

**Причина:** В таблице есть телефоны в неправильном формате

**Решение:**
```sql
-- Посмотреть проблемные телефоны
SELECT id, full_name, phone
FROM drivers
WHERE phone !~ '^\+996 \d{3} \d{3} \d{3}$';

-- Исправить их вручную
```

## ✅ ПРОВЕРКА ЧТО ВСЕ РАБОТАЕТ

После выполнения всех 3 файлов, запустите эту проверку:

```sql
-- Скопируйте и выполните в SQL Editor
SELECT
    '📊 СТАТУС ВСЕХ ИСПРАВЛЕНИЙ' as title;

-- Проверка constraints
SELECT
    'Constraints' as type,
    conname as name,
    '✅' as status
FROM pg_constraint
WHERE conname IN (
    'unique_driver_phone',
    'check_phone_format',
    'check_hire_date_not_future',
    'check_trip_date_range',
    'check_license_plate_format'
)
ORDER BY name;

-- Проверка triggers
SELECT
    'Triggers' as type,
    tgname as name,
    '✅' as status
FROM pg_trigger
WHERE tgname IN (
    'check_vehicle_availability_trigger',
    'check_driver_availability_trigger',
    'update_routes_updated_at',
    'update_trips_updated_at'
)
OR tgname LIKE 'audit_%'
ORDER BY name;

-- Проверка новых таблиц
SELECT
    'Tables' as type,
    table_name as name,
    '✅' as status
FROM information_schema.tables
WHERE table_schema = 'public'
AND table_name IN ('audit_log', 'system_settings')
ORDER BY name;

-- Проверка новых колонок
SELECT
    'Columns' as type,
    table_name || '.' || column_name as name,
    '✅' as status
FROM information_schema.columns
WHERE table_schema = 'public'
AND (
    (table_name IN ('routes', 'trips') AND column_name = 'updated_at')
    OR
    (table_name IN ('drivers', 'vehicles', 'routes', 'trips') AND column_name = 'deleted_at')
)
ORDER BY name;
```
